import {withCompany} from "@/lib/company/persistence";
import {exportRecords} from "@/lib/company/selectors";
import {operationFilters} from "@/lib/operations/filters";
import {companyDB,requestIdentity} from "@/lib/company/runtime";
import {recordKinds,salesChannels} from "@/lib/operations/types";
function cell(value:unknown){let text=String(value??"");if(/^[=+\-@\t\r]/.test(text))text="'"+text;return '"'+text.replaceAll('"','""')+'"';}
export async function GET(request:Request){const identity=requestIdentity(request);if(!identity)return Response.json({error:"Oturum açın."},{status:401});try{const params=new URL(request.url).searchParams,filters=operationFilters(params),records=await withCompany(companyDB(),identity,(state,actor)=>exportRecords(state,actor,filters,params.get("view")==="takvim"?"due":"record"));const rows=[["Kayıt no","Tür","Tarih","Kanal","Müşteri / firma","Kategori","Belge no","Açıklama","Toplam (TL)","Güncel mahsup (TL)","Güncel kalan (TL)","Durum","Personel"],...records.map(r=>[r.number,recordKinds.find(k=>k.id===r.kind)!.name,r.date,salesChannels.find(c=>c.id===r.channel)?.name??"",r.party,r.category,r.documentNo,r.description,(r.amountMinor/100).toFixed(2).replace(".",","),(r.settledMinor/100).toFixed(2).replace(".",","),(r.remainingMinor/100).toFixed(2).replace(".",","),r.status==="ACTIVE"?"Etkin":"İptal",r.createdByName])];return new Response("\ufeff"+rows.map(row=>row.map(cell).join(";")).join("\r\n"),{headers:{"Content-Type":"text/csv; charset=utf-8","Content-Disposition":`attachment; filename="sehpaciyiz-${filters.from}-${filters.to}.csv"`,"Cache-Control":"private, no-store"}});}catch{return Response.json({error:"Dışa aktarma için şirket sahibi hesabıyla giriş yapın."},{status:403});}}
